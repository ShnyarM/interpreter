function makeGrid(number size):number[][]{
    number[][] g;
    number r = 0;
    while(r < size){
        number[] row;
        number c = 0;
        while(c < size){
            row[c] = 0;
            c = c+1;
        }
        g[r] = row;
        r = r+1;
    }
    return g;
}

function render(number[][] g, number n):string{
    string out = "";
    number r = 1;
    while(r < n+1){
        number c = 1;
        while(c < n+1){
            if(g[r][c] == 1) out = out + "# ";
            else out = out + ". ";
            c = c+1;
        }
        print(out);
        out = "";
        r = r+1;
    }
    return out;
}

function step(number[][] g, number n):number[][]{
    number[][] next = makeGrid(n+2);
    number r = 0;
    while(r < n){
        number c = 0;
        while(c < n){
            number total = g[r][c]   + g[r][c+1]   + g[r][c+2]
                         + g[r+1][c] + g[r+1][c+1] + g[r+1][c+2]
                         + g[r+2][c] + g[r+2][c+1] + g[r+2][c+2];
            if(total == 3){
                next[r+1][c+1] = 1;
            } else {
                if(total == 4){
                    next[r+1][c+1] = g[r+1][c+1];
                }
            }
            c = c+1;
        }
        r = r+1;
    }
    return next;
}

number n = 10;
number[][] grid = makeGrid(n+2);

grid[1][2] = 1;
grid[2][3] = 1;
grid[3][1] = 1;
grid[3][2] = 1;
grid[3][3] = 1;


if(length(_sys_args) < 2){
  print("Please specify the number of generations the simulation should run for!");
  stop;
}

number gen = 0;
number maxGen = castToNumber(_sys_args[1]);
while(gen < maxGen){
    print("Generation " + gen);
    render(grid, n);
    print("");
    grid = step(grid, n);
    gen = gen+1;
}
