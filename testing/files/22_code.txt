function twoArrays(number[] a, number[] b, number[][] c){
    print(a[0]);
    print(b[2]);
    print(c[0][0]);
}

function getArray():number[]{
    number[] arr;
    arr[6] = 0;
    return arr;
}

function getDoubleArray(number[] inner):number[][]{
    number[][] arr;
    arr[1] = inner;
    arr[0][0] = 4;
    return arr;
}

number[] myown;
myown[2] = 6;
twoArrays(getArray(), myown, getDoubleArray(getArray()));
